% ===================== VECTEURS (suite) : simplifier avec Chasles ============
\autoCardLine{questionC}{Vecteurs}{Somme, relation de Chasles (suite)}
\vspace{6pt}
\ExempleBoxR{
  Simplifier $\Vj{AB}-\Vj{AC}-\Vj{CB}$~:\\[4pt]
  \textbf{Méthode 1} \textit{(passer aux vecteurs opposés)}
  \[
  \begin{array}{r@{\,}c@{\,}l}
  \Vj{AB}-\Vj{AC}-\Vj{CB} &=& \Vj{AB}+\Vj{CA}+\Vj{BC}\\
   &=& \Vj{AB}+\Vj{BC}+\Vj{CA}\\
   &=& \Vj{AC}+\Vj{CA}=\Vj{0}
  \end{array}
  \]
  \textbf{Méthode 2} \textit{(regrouper avec Chasles)}
  \[
  \begin{array}{r@{\,}c@{\,}l}
  \Vj{AB}-\Vj{AC}-\Vj{CB} &=& \Vj{AB}-(\Vj{AC}+\Vj{CB})\\
   &=& \Vj{AB}-\Vj{AB}\\
   &=& \Vj{0}
  \end{array}
  \]
}

% ====================== A9 -- DÉVELOPPER, FACTORISER =========================
\autoCardLine{finalC}{A9}{Développer, factoriser}
\vspace{6pt}
\ARetenirBox{
  \textit{La double distributivité a été rappelée en Série 2 -- cette
  semaine, on travaille la procédure inverse.}\\[6pt]
  \textbf{Factoriser}, c'est transformer une somme en produit. Pour tous
  réels $k$, $a$ et $b$~:
  \[ka+kb=k(a+b) \qquad ka-kb=k(a-b)\]
  On repère le \textbf{facteur commun} $k$ dans chaque terme de la somme.
}
\vspace{0pt}
\ExempleBoxR{
  Factoriser $B=7x-21$.\\[2pt]
  On repère le facteur commun~: $7$.
  \[
  \begin{array}{r@{\,}c@{\,}l}
  B &=& 7\times x-7\times3\\
   &=& 7(x-3)
  \end{array}
  \]
  \vspace{2pt}
}

\newpage

% ===================== A9 (suite) : facteur commun contenant x ===============
\autoCardLine{finalC}{A9}{Développer, factoriser (suite)}
\vspace{6pt}
\ExempleBoxR{
  Factoriser $C=3x^2+12x$.\\[2pt]
  On repère le facteur commun~: $3x$.
  \[
  \begin{array}{r@{\,}c@{\,}l}
  C &=& 3x\times x+3x\times4\\
   &=& 3x(x+4)
  \end{array}
  \]
  \vspace{2pt}
}

% ====================== A8 -- CALCUL LITTÉRAL ÉLÉMENTAIRE =====================
\autoCardLine{qcmC}{A8}{Calcul littéral élémentaire}
\vspace{6pt}
\ARetenirBox{
  \textbf{Une longueur en fonction de x}\\[4pt]
  Sur une figure codée, les morceaux qui portent le même codage ont la
  même longueur. On \textbf{additionne} tous les morceaux, puis on
  \textbf{réduit} l'expression. Si l'on connaît la longueur totale, on
  retrouve $x$ en résolvant une \textbf{équation}.
}
\vspace{0pt}
\ExempleBoxR{
  \begin{minipage}[c]{0.44\linewidth}
  Exprimer $CE$ en fonction de $x$~:
  \begin{center}
  \SegmentCode{1.6}
  \end{center}
  \end{minipage}%
  \hfill
  \begin{minipage}[c]{0.52\linewidth}
  Trois morceaux codés $\circ$ mesurent $x$, deux morceaux codés $/\!/$ mesurent $4$~:
  \[CE=x+4+x+4+x=3x+8\]
  Si $CE=23$~: $3x+8=23$, donc $3x=15$ et $x=5$.
  \end{minipage}
}
