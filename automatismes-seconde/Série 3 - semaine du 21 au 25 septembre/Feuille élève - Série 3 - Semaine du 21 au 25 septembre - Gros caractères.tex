% ============================ VECTEURS =======================================
\autoCardLine{questionC}{Vecteurs}{Somme, relation de Chasles}
\vspace{6pt}
\ARetenirBox{
  \begin{minipage}[t]{0.47\linewidth}
  \textbf{1. Somme bout à bout} (rappel Série 2)\\[4pt]
  Pour construire $\Vj{u}+\Vj{v}$, on place $\Vj{v}$ à la suite de
  $\Vj{u}$~: le vecteur somme va de l'origine de $\Vj{u}$ à l'extrémité
  de $\Vj{v}$.
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.47\linewidth}
  \textbf{2. Relation de Chasles} (nouveau)\\[4pt]
  Pour tous points $A$, $M$, $B$~:
  \[\Vj{AB}=\Vj{AM}+\Vj{MB}\]
  \end{minipage}
}
\vspace{0pt}
\ExempleBoxR{
  Simplifier $\Vj{AB}-\Vj{AC}-\Vj{CB}$~:
  \[
  \begin{array}{r@{\,}c@{\,}l}
  \Vj{AB}-\Vj{AC}-\Vj{CB} &=& \Vj{AB}-(\Vj{AC}+\Vj{CB})\\
   &=& \Vj{AB}-\Vj{AB}\\
   &=& \Vj{0}
  \end{array}
  \]
  \textit{On regroupe d'abord les deux derniers termes avec Chasles
  ($\Vj{AC}+\Vj{CB}=\Vj{AB}$), avant de soustraire.}
}

% ====================== A9 -- DÉVELOPPER, FACTORISER =========================
\autoCardLine{finalC}{A9}{Développer, factoriser}
\vspace{6pt}
\ARetenirBox{
  \textit{La double distributivité a été rappelée en Série 2 -- cette
  semaine, on travaille surtout l'opération inverse.}\\[6pt]
  \textbf{Factoriser}, c'est transformer une somme en produit. Pour tous
  réels $k$, $a$ et $b$~:
  \[ka+kb=k(a+b) \qquad ka-kb=k(a-b)\]
  On repère le \textbf{facteur commun} $k$ dans chaque terme de la somme.
}
\vspace{0pt}
\ExempleBoxR{
  Factoriser $B=7x-21$~:
  \[
  \begin{array}{r@{\,}c@{\,}l}
  B &=& 7\times x-7\times3\\
   &=& 7(x-3)
  \end{array}
  \]
  \textit{$7$ est le facteur commun à $7x$ et à $21=7\times3$.}
}

% ====================== A8 -- CALCUL LITTÉRAL ÉLÉMENTAIRE =====================
\autoCardLine{qcmC}{A8}{Calcul littéral élémentaire}
\vspace{6pt}
\ARetenirBox{
  \begin{minipage}[t]{0.47\linewidth}
  \textbf{1. Fractions avec du littéral}\\[4pt]
  On peut simplifier une écriture fractionnaire contenant une lettre
  exactement comme une fraction numérique~: on cherche un facteur
  commun au numérateur et au dénominateur.
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.47\linewidth}
  \textbf{2. Une longueur en fonction de x}\\[4pt]
  Sur une figure, une longueur totale peut se lire comme une somme de
  longueurs connues (dont certaines valent un même nombre de fois $x$).
  \end{minipage}
}
\vspace{0pt}
\ExempleBoxR{
  \begin{minipage}[t]{0.44\linewidth}
  Simplifier $\dfrac{3x+9}{3}$~:
  \[\dfrac{3x+9}{3}=\dfrac{3\times x+3\times3}{3}=\dfrac{3(x+3)}{3}=x+3\]
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.5\linewidth}
  Exprimer $AB$ en fonction de $x$~:
  \begin{center}
  \begin{tikzpicture}[scale=1.1, baseline=0pt]
    \draw[thick] (0,0) node[left]{$A$} -- (5,0) node[right]{$B$};
    \foreach \i in {0,1,2,3,4,5} \draw (\i,0.1) -- (\i,-0.1);
    \draw[decorate, decoration={brace, amplitude=5pt}, yshift=-8pt]
      (0,0) -- (2,0) node[midway, yshift=-16pt]{$x$};
    \draw[decorate, decoration={brace, amplitude=5pt}, yshift=-8pt]
      (2,0) -- (5,0) node[midway, yshift=-16pt]{$7$};
  \end{tikzpicture}
  \end{center}
  \[AB=x\times2+7\times3=2x+21\]
  \end{minipage}
}

% ============================ ALGO ============================================
\autoCardLine{qcmC}{Algo}{}
\vspace{6pt}
\ARetenirBox{
  Une fonction (Python) ou un bloc personnalisé (Scratch) peut recevoir
  \textbf{plusieurs paramètres} -- rappel Série 2~: un seul paramètre.
  En Python, on les sépare par une virgule~:
  \fbox{\texttt{def nom(param1, param2):}}. L'appel \texttt{nom(v1, v2)}
  remplace \texttt{param1} par \texttt{v1} et \texttt{param2} par
  \texttt{v2}.
}
\vspace{0pt}
\ExempleBoxR{
\begin{tcbraster}[raster columns=20, raster column skip=10pt, raster force size=false, raster valign=top, before skip=0pt, after skip=0pt]
\PanelBox[raster multicolumn=6]{\centering
  \fontsize{13}{15}\selectfont
  \begin{scratch}[scale=0.72]
    \blockinit{quand \greenflag{} est cliqué}
    \blockmoreblocks{aire \ovalnum{5} \ovalnum{3}}
  \end{scratch}
}
\PanelBox[raster multicolumn=8]{\centering
  \fontsize{13}{15}\selectfont
  \begin{scratch}[scale=0.68]
    \initmoreblocks{définir \namemoreblocks{aire \ovalmoreblocks{L} \ovalmoreblocks{l}}}
    \blocklook{dire \ovaloperator{\ovalmoreblocks{L} $\times$ \ovalmoreblocks{l}}}
  \end{scratch}
}
\PanelBox[raster multicolumn=6]{\centering\stagePanel{15}}
\end{tcbraster}
