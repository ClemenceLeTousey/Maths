\autoCardLine{questionC}{Vecteurs}{Somme, relation de Chasles}
\vspace{1pt}
\RecapPair{%
  \textbf{1. Somme bout à bout} (rappel Série 2)\\
  Pour construire $\Vj{u}+\Vj{v}$, on place $\Vj{v}$ à la suite de $\Vj{u}$~:
  le vecteur somme va de l'origine de $\Vj{u}$ à l'extrémité de $\Vj{v}$.\\[4pt]
  \textbf{2. Relation de Chasles} (nouveau)\\
  Pour tous points $A$, $M$, $B$~:
  \[\Vj{AB}=\Vj{AM}+\Vj{MB}\]
  \begin{center}\ChaslesFig{0.85}\end{center}
}{%
  Construire $\Vj{u}+\Vj{v}$ bout à bout~:
  \begin{center}
  \SommeSchemaUn{0.38}\hspace{16pt}\SommeSchemaTrois{0.38}
  \end{center}
  Simplifier $\Vj{AB}-\Vj{AC}-\Vj{CB}$~:\\[2pt]
  \textbf{Méthode 1} \textit{(passer aux vecteurs opposés)}
  \[
  \begin{array}{r@{\,}c@{\,}l}
  \Vj{AB}-\Vj{AC}-\Vj{CB} &=& \Vj{AB}+\Vj{CA}+\Vj{BC}\\
   &=& \Vj{AB}+\Vj{BC}+\Vj{CA}\\
   &=& \Vj{AC}+\Vj{CA}=\Vj{0}
  \end{array}
  \]
  \textbf{Méthode 2} \textit{(regrouper avec Chasles)}
  \[
  \begin{array}{r@{\,}c@{\,}l}
  \Vj{AB}-\Vj{AC}-\Vj{CB} &=& \Vj{AB}-(\Vj{AC}+\Vj{CB})\\
   &=& \Vj{AB}-\Vj{AB}\\
   &=& \Vj{0}
  \end{array}
  \]
}

\autoCardLine{finalC}{A9}{Développer, factoriser}
\vspace{1pt}
\RecapPair{%
  \textit{La double distributivité a été rappelée en Série 2 -- cette
  semaine, on travaille la procédure inverse.}\\[4pt]
  \textbf{Factoriser}, c'est transformer une somme en produit. Pour tous
  réels $k$, $a$ et $b$~:
  \[ka+kb=k(a+b) \qquad ka-kb=k(a-b)\]
  On repère le \textbf{facteur commun} $k$ dans chaque terme de la somme.
}{%
  Factoriser $B=7x-21$.\\[2pt]
  On repère le facteur commun~: $7$.
  \[
  \begin{array}{r@{\,}c@{\,}l}
  B &=& 7\times x-7\times3\\
   &=& 7(x-3)
  \end{array}
  \]
  \vspace{2pt}
  Factoriser $C=3x^2+12x$.\\[2pt]
  On repère le facteur commun~: $3x$.
  \[
  \begin{array}{r@{\,}c@{\,}l}
  C &=& 3x\times x+3x\times4\\
   &=& 3x(x+4)
  \end{array}
  \]
  \vspace{2pt}
}

\autoCardLine{qcmC}{A8}{Calcul littéral élémentaire}
\vspace{1pt}
\begin{tcbraster}[raster columns=2, raster column skip=8pt, raster force size=false, raster valign=top, before skip=0pt, after skip=0pt]
\ARetenirBox{\normalsize
  \textbf{Une longueur en fonction de x}\\
  Sur une figure codée, les morceaux qui portent le même codage ont la
  même longueur. On \textbf{additionne} tous les morceaux, puis on
  \textbf{réduit} l'expression.\\[4pt]
  Si l'on connaît la longueur totale, on retrouve $x$ en résolvant une
  \textbf{équation}.
}
\ExempleBoxR{\footnotesize
  Exprimer $CE$ en fonction de $x$~:
  \begin{center}
  \SegmentCode{0.8}
  \end{center}
  Trois morceaux codés $\circ$ mesurent $x$, deux morceaux codés $/\!/$ mesurent $4$~:
  \[CE=x+4+x+4+x=3x+8\]
  Si $CE=23$~: $3x+8=23$, donc $3x=15$ et $x=5$.
}
\end{tcbraster}
